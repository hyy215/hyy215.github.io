AgentScope 2.0 源码解读 · 工具集 ​

备注 · 本文的源码分析基于 AgentScope 2.0.12dev（提交 7b5b37d5）。

上一篇用 AgentState 保存了用户的翻译请求和助手的译文。要回答「上海现在下雨吗？」，模型需要上海当前的天气数据。应用把 get_current_weather 的用途和参数告诉模型，模型就可以请求查询上海的天气，再由应用执行查询。

例子 ​

这个例子用模拟数据查询上海天气。应用将天气工具注册到 Toolkit，让模型生成调用请求，再执行查询并打印结果。

python
import asyncio
import os

from agentscope.credential import DashScopeCredential
from agentscope.message import TextBlock, ToolCallBlock, UserMsg
from agentscope.model import DashScopeChatModel
from agentscope.state import AgentState
from agentscope.tool import FunctionTool, ToolChoice, Toolkit, ToolResponse


def get_current_weather(city: str) -> str:
    """查询指定城市的当前天气。

    Args:
        city: 城市名称。
    """
    weather_by_city = {"上海": "小雨，18°C", "北京": "晴，20°C"}
    return weather_by_city.get(city, "暂无天气数据")


async def main() -> None:
    model = DashScopeChatModel(
        credential=DashScopeCredential(api_key=os.environ["DASHSCOPE_API_KEY"]),
        model="qwen-plus",
        stream=False,
        parameters=DashScopeChatModel.Parameters(thinking_enable=False),
    )

    message = UserMsg(name="reader", content="上海现在下雨吗？")

    tool = FunctionTool(get_current_weather)
    toolkit = Toolkit(tools=[tool])
    schemas = await toolkit.get_tool_schemas()

    response = await model(
        [message],
        tools=schemas,
        tool_choice=ToolChoice(mode=tool.name),
    )

    state = AgentState()
    for block in response.content:
        if not isinstance(block, ToolCallBlock):
            continue

        print(f"工具名称：{block.name}")
        print(f"调用参数：{block.input}")

        async for result in toolkit.call_tool(block, state):
            if isinstance(result, ToolResponse):
                for item in result.content:
                    if isinstance(item, TextBlock):
                        print(f"执行结果：{item.text}")


asyncio.run(main())

定义工具 ​

ToolBase 工具基类 ​

ToolBase 是 AgentScope 的工具抽象基类。内置工具、FunctionTool 和 MCPTool 都继承自它，通过同一套接口接入 Toolkit。

主要属性

  • name 指定工具名称。
  • description 描述工具用途。
  • input_schema 定义参数结构和约束。
  • is_concurrency_safe 标记工具是否支持并发调用。
  • is_read_only 标记工具是否只读。

主要方法

  • call() 执行工具逻辑，返回一个 ToolChunk，也可以通过异步生成器逐段产出 ToolChunk。
  • check_permissions() 返回本次调用的权限判断，供 Agent 决策。
python

    is_read_only: bool

        *args: Any,
        **kwargs: Any,
    ) -> ToolChunk | AsyncGenerator[ToolChunk, None]:

ToolChunk 工具输出块 ​

ToolChunk 表示工具执行产生的一段输出。

  • content 包含本段产出的 TextBlock 或 DataBlock。
  • state 默认为 RUNNING（执行中），还可以是 SUCCESS（成功）、ERROR（出错）、DENIED（拒绝）或 INTERRUPTED（中断）。
  • is_last 标记本段是否是最后一段，默认是 True。例如，MCPTool 关闭进度输出时会用它选出最终块。
python
# ToolChunk 字段节选，省略元数据和 ID。
class ToolChunk(BaseModel):
    content: List[TextBlock | DataBlock]
    state: ToolResultState = ToolResultState.RUNNING
    is_last: bool = True

例如，Read 工具继承 ToolBase。它的 call() 先校验路径，再按文件扩展名选择读取方法，返回 ToolChunk。

python
# Read 类节选，保留工具声明和按扩展名分派的逻辑。
class Read(ToolBase):
    name: str = "Read"
    input_schema: dict[str, Any] = _ReadParams.model_json_schema()
    is_read_only: bool = True

    async def call(  # type: ignore[override]
        self,
        file_path: str,
        offset: int = 1,
        limit: int = 2000,
        pages: str | None = None,
        _agent_state: AgentState | None = None,
    ) -> ToolChunk:
        # 省略绝对路径、文件是否存在及是否为目录的检查。
        ext = os.path.splitext(self._backend.basename(file_path))[1].lower()

        if ext == ".pdf":
            # 省略 PDF 读取失败时的处理。
            return await self._read_pdf(file_path, pages)
        if ext in _IMAGE_EXTENSIONS:
            return await self._read_image_file(file_path, ext)

        # 其余文件按普通文本读取。
        return await self._read_text_file(
            file_path,
            offset,
            limit,
            _agent_state,
        )

用 FunctionTool 自定义 Python 工具 ​

FunctionTool 把普通 Python 函数包装成 ToolBase 实例。

  • 构造时，__init__() 准备工具名称、描述和参数 schema。
  • 执行时，call() 调用原函数，并将结果转换为 ToolChunk。
python
# FunctionTool 类节选，省略类级标记和文档字符串。
class FunctionTool(ToolBase):
    def __init__(
        self,
        func: Function,
        name: str | None = None,
        description: str | None = None,
        # 可显式传入 JSON Schema 字典或 Pydantic 模型类。
        input_schema: dict | type[BaseModel] | None = None,
        is_concurrency_safe: bool = True,
        is_read_only: bool = False,
        is_state_injected: bool = False,
        middlewares: list[ToolMiddlewareBase] | None = None,
        permission: PermissionDecision | None = None,
    ) -> None:
        # 省略配置初始化。默认从函数名、文档字符串和类型标注推导工具声明。
        # 原函数和权限配置分别保存到 _func 和 _permission。
        ...

    async def check_permissions(
        self,
        *_args: Any,
        **_kwargs: Any,
    ) -> PermissionDecision:
        if self._permission is not None:
            return self._permission

        return PermissionDecision(
            behavior=PermissionBehavior.ASK,
            message="Custom function tools must be explicitly allowed "
            "by the user.",
        )

    async def call(
        self,
        **kwargs: Any,
    ) -> ToolChunk | AsyncGenerator[ToolChunk, None]:
        result = self._func(**kwargs)
        if inspect.isawaitable(result):
            result = await result
        # 省略同步和异步生成器逐块转换的分支。
        return self._convert_func_result_to_chunk(result)

    @staticmethod
    def _convert_func_result_to_chunk(
        result: Any,
    ) -> ToolChunk:
        if isinstance(result, ToolChunk):
            return result
        if isinstance(result, str):
            text = result
        else:
            try:
                text = json.dumps(result, ensure_ascii=False)
            except (TypeError, ValueError):
                text = str(result)
        return ToolChunk(
            content=[TextBlock(text=text)],
            state=ToolResultState.RUNNING,
        )

MCP 工具 ​

MCPClient 管理与 MCP 服务端的连接。它提供的 list_tools() 接口分两步生成并返回 MCPTool 实例列表。

  1. 调用 list_raw_tools() 获取服务端的可用工具定义。
  2. 按工具名逐个调用 get_tool()，将每个工具包装成 MCPTool 实例。

class MCPClient(BaseModel):

    is_stateful: bool
    mcp_config: StdioMCPConfig | HttpMCPConfig

    async def connect(self) -> None:
        # 建立有状态会话，省略传输和任务管理。
        ...

    async def close(self, ignore_errors: bool = True) -> None:
        # 省略会话关闭和清理。
        ...

    async def list_raw_tools(self) -> list[mcp.types.Tool]:
        # 返回原始工具定义。省略工具发现和启停名单过滤。
        ...

    async def list_tools(self) -> list[ToolBase]:
        raw_tools = await self.list_raw_tools()
        return [await self.get_tool(_.name) for _ in raw_tools]

    async def get_tool(
        self,
        name: str,

                timeout=self.execution_timeout,
                stream_progress=self.stream_progress,

        else:
            self._validate_connection()
            return MCPTool(
                mcp_name=self.name,
                tool=target_tool,
                session=self._client,
                timeout=self.execution_timeout,
                stream_progress=self.stream_progress,
            )

MCPTool ​

MCPTool 把服务端的 MCP 工具包装成 ToolBase 实例，向模型提供带 mcp__ 前缀的工具名。

  • check_permissions() 根据服务端声明的 read_only_hint 给出工具级权限判断。
  • call() 使用原始工具名调用 MCP 服务端，并将返回结果转换成 ToolChunk。

    is_state_injected: bool = False

        sanitized_tool = re.sub(r"[^a-zA-Z0-9_-]", "x", tool.name)
        self.name = f"mcp__{mcp_name}__{sanitized_tool}"
        self.is_read_only = False
        if tool.annotations:
            self.is_read_only = tool.annotations.read_only_hint or False
        # 省略参数 schema、会话和进度配置的初始化。

    async def check_permissions(
        self,
        *_args: Any,
        **_kwargs: Any,
    ) -> PermissionDecision:
        if self.is_read_only:
            return PermissionDecision(
                behavior=PermissionBehavior.ALLOW,
                message="This is a read-only MCP tool. Allowing execution.",
            )
        return PermissionDecision(
            behavior=PermissionBehavior.ASK,
            message="MCP tools must be explicitly allowed by the user.",
        )

    async def call(self, **kwargs: Any) -> ToolChunk | AsyncGenerator[ToolChunk, None]:
        if self.stream_progress:
            return self._stream_call(kwargs)

        async for chunk in self._stream_call(kwargs):
            if chunk.is_last:
                final_chunk = chunk

        return final_chunk

        self,

        async def _invoke() -> mcp.types.CallToolResult:
            return await client.call_tool(self._tool.name, arguments=kwargs)

        call_task = asyncio.create_task(_invoke())
        # 省略等待请求期间的进度转发。
        result = await call_task

        content = self._convert_mcp_content_to_blocks(result.content)

        yield ToolChunk(
            content=content,
            state=(
                ToolResultState.ERROR
                if result.is_error
                else ToolResultState.RUNNING

            is_last=True,

    def _convert_mcp_content_to_blocks(
        mcp_content_blocks: list,
    ) -> list[TextBlock | DataBlock]:

中间件按注册顺序从外到内逐层调用，每层通过 next_handler(**input_kwargs) 把参数传给下一层，最后由框架调用工具的 call() 方法。

python
# ToolBase.__call__() 节选。
async def __call__(
    self,
    *args: Any,
    **kwargs: Any,
) -> ToolChunk | AsyncGenerator[ToolChunk, None]:
    # 省略位置参数校验。
    middlewares = getattr(self, "_middlewares", [])
    # 没有中间件时直接执行工具。
    if not middlewares:
        if inspect.isasyncgenfunction(self.call):
            return self.call(**kwargs)
        return await self.call(**kwargs)

    async def execute_chain(
        index: int = 0,
        **chain_kwargs: Any,
    ) -> AsyncGenerator[ToolChunk, None]:
        if index >= len(middlewares):
            # 最内层执行工具。保留单个 ToolChunk 的路径，省略流式返回的处理。
            result = await self.call(**chain_kwargs)
            yield result
        else:
            mw = middlewares[index]
            input_kwargs = dict(chain_kwargs)

            async def next_handler(
                **kw: Any,
            ) -> AsyncGenerator[ToolChunk, None]:
                async for chunk in execute_chain(index + 1, **kw):
                    yield chunk

            async for chunk in mw.on_tool_call(
                tool=self,
                input_kwargs=input_kwargs,
                next_handler=next_handler,
            ):
                yield chunk

    # 返回从第一层开始的结果流，调用方迭代时运行中间件。
    return execute_chain(**kwargs)

用 ToolGroup 定义工具组 ​

ToolGroup 把同一任务需要的工具、MCP 客户端和 Skill 组织到一组。

  • name 标识工具组，选用工具组时传入这个名称。

注册工具 ​

收集工具 ​

Toolkit 通过 _get_available_tools() 方法收集 basic 组和指定组（groups）里的工具。

  • 普通工具直接取自 group.tools。
  • MCP 工具通过 client.list_tools() 获取。
  • 内置的 Skill 工具在 _get_available_skills(groups) 查到可用 Skill 后才加入。加载和读取流程见下文「Skill 的注册与读取」。

最终返回一个工具字典，键是工具名，值是封装该工具的 RegisteredTool 实例。

python
# Toolkit._get_available_tools() 节选。
async def _get_available_tools(
    self,
    groups: list[str] | None,
) -> dict[str, RegisteredTool]:
    available_tools = {}
    skills = await self._get_available_skills(groups)
    if len(skills):
        available_tools[
            self.builtin_skill_viewer.tool.name
        ] = self.builtin_skill_viewer

    # 省略元工具的装载。
    groups_filter = ["basic"] + (groups or [])
    for group in self.tool_groups:
        if group.name not in groups_filter:
            continue

        cache_tools = []
        for tool in group.tools:
            cache_tools.append(tool)

        for client in group.mcps:
            # 省略 MCP 工具查询失败时的跳过和日志处理。
            cache_tools.extend(await client.list_tools())

        for tool in cache_tools:
            # 省略重名告警，后收集的同名工具覆盖前一个。
            available_tools[tool.name] = RegisteredTool(
                tool=tool,
                group=group.name,
            )

    return available_tools

async def call_tool(
    self,
    tool_call: ToolCallBlock,
    state: AgentState,

        state.tool_context.activated_groups,
    )

ToolResponse 工具执行结果 ​

工具调用过程中，Toolkit 逐块产出 ToolChunk，并用 ToolResponse.append_chunk() 汇总结果。

  1. 追加内容，按 ID 拼接 Base64 片段。
  2. 更新状态和元数据。
  3. 合并相邻的文本块。
python
# ToolResponse 节选，保留字段和 append_chunk() 的内容汇总逻辑。
class ToolResponse(BaseModel):
    content: List[TextBlock | DataBlock] = Field(default_factory=list)
    state: Literal[  # type: ignore[valid-type]
        ToolResultState.ERROR,
        ToolResultState.DENIED,
        ToolResultState.INTERRUPTED,
        ToolResultState.SUCCESS,
    ] = ToolResultState.SUCCESS
    metadata: dict = Field(default_factory=dict)
    id: str = Field(default_factory=_generate_id)

    def append_chunk(self, chunk: ToolChunk) -> Self:
        # 省略为已有 Base64 数据块建立索引的逻辑。
        for block in chunk.content:
            if block.type == "text" or block.source.type == "url":
                self.content.append(block.model_copy(deep=True))
                continue
            # 省略 Base64 数据块的追加和按 ID 拼接。

        # 省略状态和元数据的更新。
        # 合并相邻的文本块。
        merged_content: List[TextBlock | DataBlock] = []
        for block in self.content:
            if isinstance(block, TextBlock) and merged_content:
                last_block = merged_content[-1]
                if isinstance(last_block, TextBlock):
                    last_block.text += block.text
                else:
                    merged_content.append(block)
            else:
                merged_content.append(block)

        self.content = merged_content

        return self

LocalSkillLoader 对外提供异步方法 list_skills()，用于查找并解析本地目录中的 SKILL.md，缓存生成的 Skill 对象并返回列表。

python
# LocalSkillLoader 节选，省略部分校验、异常处理和日志。
class LocalSkillLoader(SkillLoaderBase):
    def __init__(self, directory: str, scan_subdir: bool = False) -> None:
        self.directory = _normalize_local_path(directory)
        self.scan_subdir = scan_subdir
        self._cache: dict[str, Skill] = {}

    async def list_skills(self) -> list[Skill]:
        # 省略 _find_skill_dirs() 的定义。
        skill_dirs = await asyncio.to_thread(_find_skill_dirs)
        tasks = [
            self._load_single_skill(skill_dir) for skill_dir in skill_dirs
        ]
        results = await asyncio.gather(*tasks, return_exceptions=True)

        skills: list = []
        for result in results:
            if isinstance(result, Exception):
                continue
            elif result is not None:
                skills.append(result)

        return skills

    async def _load_single_skill(self, skill_root: str) -> Skill | None:
        skill_md_path = os.path.join(skill_root, "SKILL.md")
        updated_at = await aiofiles.ospath.getmtime(skill_md_path)
        if skill_root in self._cache:
            cached_skill = self._cache[skill_root]
            if cached_skill.updated_at == updated_at:
                return cached_skill

        async with aiofiles.open(
            skill_md_path,
            "r",
            encoding="utf-8-sig",
        ) as f:
            content_str = await f.read()
            content = frontmatter.loads(content_str)

        name = content.get("name")
        description = content.get("description")
        skill = Skill(
            name=str(name),
            description=str(description),
            dir=skill_root,
            markdown=content.content,
            updated_at=updated_at,
        )
        self._cache[skill_root] = skill

        return skill

async def _get_available_skills(

    groups_filter = ["basic"] + (groups or [])
